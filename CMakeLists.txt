# Local exercises may use the downloaded video/reference corpus. They remain
# opt-in executables and are not part of the product build or CTest suite.
function(tiger_exercise name)
 add_executable(${name} EXCLUDE_FROM_ALL tests/${name}.cpp)
 target_include_directories(${name} PRIVATE include)
 target_link_libraries(${name} PRIVATE nlohmann_json::nlohmann_json ${ARGN})
endfunction()
function(tiger_test name)
 add_executable(${name} tests/${name}.cpp)
 target_include_directories(${name} PRIVATE include)
 target_link_libraries(${name} PRIVATE nlohmann_json::nlohmann_json ${ARGN})
 add_test(NAME ${name} COMMAND ${name})
 set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
